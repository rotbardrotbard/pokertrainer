    25→/** Позиции за столом на N игроков (порядок действия на префлопе). */
    26→export function seatsForTable(players: number): string[] {
    27→  if (players === 2) return ['BTN', 'BB'] // хедз-ап: BTN — он же SB
    28→  const early = players - 3
    29→  const seats: string[] = []
    30→  const earlyNames: Record<number, string[]> = {
    31→    1: ['CO'],
    32→    2: ['MP', 'CO'],
    33→    3: ['UTG', 'MP', 'CO'],
    34→    4: ['UTG', 'MP', 'MP2', 'CO'],
    35→    5: ['UTG', 'MP', 'MP2', 'HJ', 'CO'],
    36→    6: ['UTG', 'UTG1', 'MP', 'MP2', 'HJ', 'CO'],
    37→  }
    38→  seats.push(...(earlyNames[early] ?? []))
    39→  seats.push('BTN', 'SB', 'BB')
    40→  return seats
    41→}

    43→/** Базовая позиция для чартов (MP2/HJ → MP и т.п.). */
    44→export function basePos(seat: string): Position {
    45→  if (seat.startsWith('UTG')) return 'UTG'
    46→  if (seat.startsWith('MP') || seat === 'HJ') return 'MP'
    47→  if (seat === 'CO') return 'CO'
    48→  if (seat.startsWith('BTN')) return 'BTN'
    49→  if (seat.startsWith('SB')) return 'SB'
    50→  return 'BB'
    51→}

    71→export interface HoldemPreflop {
    72→  raise: Set<string> // открытие: рейз; vs рейза: 3бет
    73→  call: Set<string> // vs рейза: колл (пусто для открытия)
    74→  raisePct: number
    75→  callPct: number
    76→  mode: 'open' | 'vsraise'
    77→}

    79→/**
    80→ * Чарт префлопа для позиции, размера стола и ситуации.
    81→ * В ситуации «против рейза» ширина защиты зависит от позиции рейзера:
    82→ * BB против BTN/SB защищается в 1,5–2 раза шире, чем против UTG.
    83→ */
    84→export function holdemPreflop(pos: Position, players: number, situation: 'open' | 'vsraise', callers: number, raiser?: Position): HoldemPreflop {
    85→  if (situation === 'open') {
    86→    const set = openRangeClassesFor(pos, players)
    87→    return { raise: set, call: new Set(), raisePct: rangePercent(set), callPct: 0, mode: 'open' }
    88→  }
    89→  const rp = raiser && raiser !== pos ? raiser : DEFAULT_RAISER[pos] ?? 'CO'
    90→  const multiFactor = Math.max(0.45, 1 - 0.1 * callers)
    91→  const chart = vsRaiseCharts(pos, rp, players, multiFactor)
    92→  if (!chart) {
    93→    return { raise: new Set(), call: new Set(), raisePct: 0, callPct: 0, mode: 'vsraise' }
    94→  }
    95→  return { raise: chart.threeBet, call: chart.call, raisePct: chart.threeBetPct, callPct: chart.callPct, mode: 'vsraise' }
    96→}
    97→
    98→export function holdemAction(cls: string, chart: HoldemPreflop): 'raise' | 'call' | 'fold' {
    99→  if (chart.raise.has(cls)) return 'raise'
   100→  if (chart.call.has(cls)) return 'call'
   101→  return 'fold'
   102→}

   150→/** Множитель ширины защиты от позиции рейзера: против поздних открытий защищаются шире. */
   151→const RAISER_FACTOR: Record<Position, number> = { UTG: 0.8, MP: 0.9, CO: 1.0, BTN: 1.25, SB: 1.35, BB: 1.0 }

   153→export interface OmahaPreflop {
   154→  mode: 'open' | 'vsraise'
   155→  openPercent: number // для открытия: ширина диапазона рейза
   156→  threeBetPercent: number // для vs-рейза
   157→  callPercent: number
   158→  openThreshold: number
   159→  threeBetThreshold: number
   160→  callThreshold: number
   161→}
   162→
   163→export function omahaPreflop(game: 'omaha4' | 'omaha5', pos: Position, players: number, situation: 'open' | 'vsraise', callers: number, raiser?: Position): OmahaPreflop {
   164→  const sizeFactor = SIZE_DEF_FACTOR[players] ?? 1
   165→  if (situation === 'open') {
   166→    const percent = clamp(PLO_OPEN_SIZE[game][players]?.[pos] ?? PLO_OPEN_SIZE[game][6][pos], 5, 90)
   167→    return {
   168→      mode: 'open',
   169→      openPercent: percent,
   170→      threeBetPercent: 0,
   171→      callPercent: 0,
   172→      openThreshold: omahaRangeThreshold(game, percent),
   173→      threeBetThreshold: 0,
   174→      callThreshold: 0,
   175→    }
   176→  }
   177→  const rp = raiser && raiser !== pos ? raiser : DEFAULT_RAISER[pos] ?? 'CO'
   178→  const rf = RAISER_FACTOR[rp]
   179→  const multiFactor = Math.max(0.45, 1 - 0.1 * callers)
   180→  const bbBonus = pos === 'BB' ? 1.5 : 1
   181→  const base = PLO_VSRAISE_POS[game][pos]
   182→  const threeBetPercent = clamp(base.threeBet * (1 + (rf - 1) / 2) * sizeFactor, 3, 18)
   183→  const callPercent = clamp(base.call * rf * sizeFactor * multiFactor * bbBonus, 5, 62)
   184→  return {
   185→    mode: 'vsraise',
   186→    openPercent: 0,
   187→    threeBetPercent,
   188→    callPercent,
   189→    openThreshold: 0,
   190→    threeBetThreshold: omahaRangeThreshold(game, threeBetPercent),
   191→    callThreshold: omahaRangeThreshold(game, callPercent),
   192→  }
   193→}
   194→
   195→export function omahaAction(game: 'omaha4' | 'omaha5', hand: Card[], chart: OmahaPreflop, situation: 'open' | 'vsraise'): 'raise' | 'call' | 'fold' {
   196→  if (situation === 'open') {
   197→    return handInRange(game, hand, chart.openPercent) ? 'raise' : 'fold'
   198→  }
   199→  if (handInRange(game, hand, chart.threeBetPercent)) return 'raise'
   200→  if (handInRange(game, hand, chart.callPercent)) return 'call'
   201→  return 'fold'
   202→}

   220→/** Анализ текстуры доски: масть, спаренность, связанность, высота. */
   221→export function analyzeBoardTexture(board: Card[]): BoardTexture {
   222→  if (board.length < 3) {
   223→    return {
   224→      label: 'Текстура не задана', note: 'Выберите предустановку доски, чтобы учесть текстуру.',
   225→      paired: false, trips: false, monotone: false, twoTone: false, rainbow: true,
   226→      connected: false, straightPossible: false, high: false, empty: true,
   227→    }
   228→  }
   229→  const rankCnt = new Map<number, number>()
   230→  const suitCnt = [0, 0, 0, 0]
   231→  for (const c of board) {
   232→    rankCnt.set(c.r, (rankCnt.get(c.r) || 0) + 1)
   233→    suitCnt[c.s]++
   234→  }
   235→  const counts = [...rankCnt.values()]
   236→  const paired = counts.some((v) => v >= 2)
   237→  const trips = counts.some((v) => v >= 3)
   238→  const monotone = board.every((c) => c.s === board[0].s)
   239→  const distinctSuits = suitCnt.filter((n) => n > 0).length
   240→  const twoTone = !monotone && distinctSuits === 2
   241→  const rainbow = !monotone && !twoTone
   242→  const distinct = [...rankCnt.keys()].sort((a, b) => a - b)
   243→  const extended = [...distinct]
   244→  if (rankCnt.has(14)) extended.push(1)
   245→  const seq = [...new Set(extended)].sort((a, b) => a - b)
   246→  let straightPossible = false
   247→  for (let i = 0; i + 2 < seq.length; i++) {
   248→    if (seq[i + 2] - seq[i] === 2) straightPossible = true
   249→  }
   250→  const span = distinct[distinct.length - 1] - distinct[0]
   251→  const connected = distinct.length >= 3 && span <= 5 && !paired
   252→  const high = Math.max(...distinct) >= 12
   253→
   254→  const parts: string[] = []
   255→  if (monotone) parts.push('моно-масть')
   256→  else if (twoTone) parts.push('двухмастная')
   257→  else parts.push('радуга')
   258→  if (paired) parts.push(trips ? 'спаренная' : 'спаренная')
   259→  if (straightPossible) parts.push('стрит-возможен')
   260→  const label = parts.join(' · ')
   261→
   262→  const notes: string[] = []
   263→  if (monotone) notes.push('одна масть на доске: блефам нечего усиливать, играют только натс-флеши и руки с тузом масти')
   264→  else if (twoTone) notes.push('масть на доске: флеш-дро живы, полу-блефы сохраняют эквити')
   265→  else notes.push('без масти: блефы работают лучше, слабые готовые руки коллируют реже')
   266→  if (paired) notes.push('пара на доске: у оппонентов меньше комбинаций, ставки можно делать дешевле')
   267→  if (straightPossible) notes.push('доска связана: сильные руки защищают банк от стрит-дро')
   268→  if (!straightPossible && !paired && rainbow) notes.push('сухая доска: чаще блеф и мелкие банки, границы коллов ниже')
   269→
   270→  return {
   271→    label,
   272→    note: notes.join('; '),
   273→    paired, trips, monotone, twoTone, rainbow, connected, straightPossible, high, empty: false,
   274→  }
   275→}

   574→function positionNote(seat: string): string {

   577→  if (p === 'SB' || p === 'BB') return 'Вы без позиции: играйте пассивнее и тайтовее, выбирайте руки, которые хорошо попадают в доску.'
   578→  return 'Средняя позиция: умеренный стиль — вэлью с сильных рук, быстрые фолды маргинальных.'
   579→}

   588→  if (opponents === 2) {

   594→function tierTextureNote(t: PlanTierBase, texture: BoardTexture, opponents: number): string {
   595→  if (texture.empty) return ''
   596→  const parts: string[] = []
   597→  if (texture.monotone) {
   598→    parts.push(t.bluffTier
   599→      ? 'на монотонной доске у блефов нет флеш-дро — блефуйте заметно реже'
   600→      : 'на монотонной доске ставки мельче: рейзы может позволить только натс-флеш')
   601→  }
   602→  if (texture.twoTone && t.bluffTier) parts.push('у полу-блефов есть флеш-дро — бет или колл сохраняют эквити')
   603→  if (texture.paired) parts.push('спаренная доска — ставки дешевле, границы коллов ниже')
   604→  if (texture.straightPossible && !t.bluffTier) parts.push('связанная доска — размер ставки чуть больше для защиты от дро')
   605→  if (texture.rainbow && !texture.paired && !texture.straightPossible && t.bluffTier) {
   606→    parts.push(opponents <= 1 ? 'сухая доска — блеф с хорошим фолд-эквити' : 'сухая доска в мультпоте — блефы почти не работают')
   607→  }
   608→  if (texture.high && t.action === 'fold') parts.push('старшие карты на доске бьют вторые пары — не цепляйтесь')
   609→  return parts.join('; ')
   610→}

   612→function planOneTier(t: PlanTierBase, texture: BoardTexture, opponents: number, facingPct: number | null): TierPlan {
   613→  let betFreq = t.betFreq
   614→  let size = t.size
   615→  const isNut = t.foldAbove >= 999
   616→
   617→  // текстура доски
   618→  if (texture.monotone) {
   619→    size = Math.min(size, 50)
   620→    if (t.bluffTier) betFreq *= 0.6
   621→    else if (!isNut) betFreq *= 0.85
   622→  }
   623→  if (texture.paired) size = Math.min(size, 40)
   624→  if (texture.straightPossible && !t.bluffTier && !isNut) size = Math.min(90, size + 10)
   625→  if (texture.rainbow && !texture.paired && !texture.straightPossible && t.bluffTier) betFreq = Math.min(80, betFreq * 1.3)
   626→
   627→  // мультпот
   628→  if (opponents >= 3) {
   629→    if (isNut) betFreq = Math.max(betFreq, 85)
   630→    if (t.bluffTier) betFreq *= 0.4
   631→    size = Math.min(size, 66)
   632→  }
   633→  if (opponents >= 5 && t.bluffTier) betFreq = 0
   634→  betFreq = Math.round(clamp(betFreq, 0, 95))
   635→  size = Math.round(size)
   636→
   637→  // смешанные частоты на инициативе
   638→  const mix: MixItem[] = []
   639→  if (betFreq <= 0) {
   640→    mix.push({ action: 'check', freq: 100, label: 'чек 100%' })
   641→  } else {
   642→    mix.push({ action: 'bet', freq: betFreq, label: `бет ${size}% банка — ${betFreq}%` })
   643→    if (betFreq < 100) mix.push({ action: 'check', freq: 100 - betFreq, label: `чек ${100 - betFreq}%` })
   644→  }
   645→  const line = betFreq >= 85
   646→    ? `бет ${size}% банка (изредка чек)`
   647→    : betFreq <= 0
   648→      ? 'чек (блефы почти не работают)'
   649→      : `микс: бет ${size}% банка в ${betFreq}% случаев / чек в ${100 - betFreq}%`
   650→
   651→  const sizeParts: string[] = []
   652→  if (texture.monotone) sizeParts.push('на монотонной доске не крупнее 50% банка')
   653→  if (texture.paired) sizeParts.push('спаренная доска — размер поменьше')
   654→  if (texture.straightPossible && !t.bluffTier && !isNut) sizeParts.push('связанная доска — размер побольше для защиты')
   655→  const sizeNote = sizeParts.length > 0 ? `Размер: ${size}% банка (${sizeParts.join('; ')}).` : `Размер: ${size}% банка.`
   656→
   657→  // границы колл/фолд против выбранного размера ставки
   658→  const multiFactor = opponents >= 4 ? 0.7 : opponents === 3 ? 0.85 : 1
   659→  const callUpTo = Math.round(t.callUpTo * multiFactor)
   660→  let verdict: FacingVerdict | null = null
   661→  if (facingPct != null) {
   662→    if (isNut) {
   663→      verdict = { verdict: 'raise', note: `${t.faceText} — рейз при любой ставке оппонента` }
   664→    } else if (facingPct <= callUpTo && facingPct <= t.foldAbove) {
   665→      verdict = { verdict: 'call', note: `колл: ставка ${facingPct}% банка в пределах границы ~${callUpTo}% сходится по оддсам` }
   666→    } else if (facingPct > t.foldAbove) {
   667→      verdict = { verdict: 'fold', note: `фолд: ставка ${facingPct}% банка выше границы ~${t.foldAbove > 900 ? 'любой' : `${t.foldAbove}%`} — не сходится` }
   668→    } else {
   669→      verdict = { verdict: 'mix', note: `пограничная ставка: микс колл/фолд${opponents >= 2 ? ', в мультпоте чаще фолд' : ''}` }
   670→    }
   671→  }
   672→  const vsBetNote = `колл до ~${callUpTo}% банка${t.foldAbove < 999 ? `, против ~${t.foldAbove}%+ — фолд` : ', вместо фолда — рейз'}`
   673→
   674→  return {
   675→    tier: t.tier,
   676→    examples: t.examples,
   677→    action: t.action,
   678→    line,
   679→    mix,
   680→    sizeNote,
   681→    vsBetNote,
   682→    verdict,
   683→    textureNote: tierTextureNote(t, texture, opponents),
   684→  }
   685→}

   687→/**
   688→ * Постфлоп-план: тиры силы с размерами ставок, смешанными частотами,
   689→ * учётом текстуры доски и границами колл/фолд против ставки заданного размера.
   690→ */
   691→export function postflopPlan(
   692→  game: GameKind,
   693→  street: Exclude<Street, 'preflop'>,
   694→  board: Card[],
   695→  opponents: number,
   696→  seat: string,
   697→  facingPct: number | null,
   698→): PostflopPlan {
   699→  const texture = analyzeBoardTexture(board)
   700→  const tiers = GUIDES[`${game}:${street}`] ?? HOLDEM_FLOP
   701→  const omahaNote = game !== 'holdem'
   702→    ? 'Особенность Омахи: сильные готовые руки появляются у многих игроков, поэтому играют только натсы, а «почти натсы» играют пассивно.'
   703→    : 'Особенность Холдема: одна пара часто достаточна против 1 оппонента, но слаба в мультпотах.'
   704→  return {
   705→    intro: `${STREET_NAMES_RU[street]}, ${opponents} ${opponents === 1 ? 'оппонент' : 'оппонентов'} в банке. Ориентир по силе руки:`,
   706→    gameNote: omahaNote,
   707→    positionNote: positionNote(seat),
   708→    multiwayNote: multiwayNote(opponents),
   709→    textureLabel: texture.empty ? texture.label : `Доска: ${texture.label}`,
   710→    textureNote: texture.note,
   711→    facingLabel: facingPct != null ? `Оппонент поставил ${facingPct}% банка` : null,
   712→    tiers: tiers.map((t) => planOneTier(t, texture, opponents, facingPct)),
   713→  }
   714→}

   716→/**
   717→ * Ориентир из GTO-чарта для Омахи по вердикту решения:
   718→ * raise → верхний тир (натсы), call → средний, fold → нижний.
   719→ * Используется симулятором раздач для связи с разделом GTO.
   720→ */
   721→export function omahaEquityTier(game: 'omaha4' | 'omaha5', street: Exclude<Street, 'preflop'>, verdict: 'raise' | 'call' | 'fold'): PostflopTier {
   722→  const tiers = GUIDES[`${game}:${street}`] ?? OMAHA_FLOP
   723→  const pick = verdict === 'raise' ? tiers[0] : verdict === 'call' ? tiers[1] : tiers[tiers.length - 1]
   724→  return { tier: pick.tier, examples: pick.examples, aggressive: pick.aggrText, facingBet: pick.faceText, action: pick.action }
   725→}

   729→export function preflopNotes(game: GameKind, pos: Position, situation: 'open' | 'vsraise', callers: number, raiser?: Position, players = 6): string[] {
   730→  const notes: string[] = []
   731→  if (game === 'holdem') {
   732→    if (situation === 'open') {
   733→      notes.push(`Стол ${players}-max: открытие с ${pos} — топ ~${openPercentFor(pos, players).toFixed(0)}% рук; отдельный чарт для каждого размера стола.`)
   734→      notes.push('Открытие рейзом 2–3bb: чем раньше позиция, тем тайтовее диапазон.')
   735→      if (pos === 'BTN') notes.push('С баттона открывается самый широкий диапазон — вы играете в позиции на всех блайндах.')
   736→      if (pos === 'UTG') notes.push('С UTG после вас действует много игроков: нужна сила руки, а не потенциал.')
   737→      if (pos === 'SB') notes.push('С SB против одного BB можно открывать шире, но без позиции постфлоп сложнее.')
   738→    } else {
   739→      const rp = raiser && raiser !== pos ? raiser : DEFAULT_RAISER[pos] ?? 'CO'
   740→      const chart = holdemPreflop(pos, players, 'vsraise', callers, rp)
   741→      notes.push(`Против открытия с ${rp} за столом ${players}-max: 3бет ≈ топ ${chart.raisePct.toFixed(0)}% рук, колл ≈ топ ${chart.callPct.toFixed(0)}%.`)
   742→      notes.push('Ширина защиты зависит от позиции рейзера: BB против BTN/SB коллирует примерно в 1,5–2 раза больше рук, чем против UTG.')
   743→      notes.push('Против рейза: премиум — 3бет, играбельные — колл, остальное — фолд.')
   744→      if (callers > 0) notes.push(`Рейз уже поддержали ${callers} ${callers === 1 ? 'игрок' : 'игрока'}: коллы требуют более сильных рук, мультпот — для натс-потенциала.`)
   745→      if (pos === 'BB') notes.push('BB защищается шире всех — колл со скидкой блайнда.')
   746→      if (pos === 'BTN') notes.push('С баттона против рейза много коллов в позиции и 3беты с премиумом.')
   747→    }
   748→  } else {
   749→    if (situation === 'open') {
   750→      const pct = omahaPreflop(game as 'omaha4' | 'omaha5', pos, players, 'open', 0).openPercent
   751→      notes.push(`Стол ${players}-max: открытие с ${pos} — топ ~${pct.toFixed(0)}% рук; отдельный чарт для каждого размера стола.`)
   752→      notes.push('В Омахе открывайте координированные руки: тузы, мастевость и связки вместе сильнее, чем по отдельности.')
   753→      if (pos === 'UTG') notes.push('С UTG — тайтово: двойномастные тузы, бродвей-связки, высокие пары.')
   754→      if (pos === 'BTN') notes.push('С баттона — самый широкий диапазон: все двойномастные, связки, пары и A-хай руки.')
   755→      if (pos === 'SB') notes.push('С SB открывайте чуть тайтовее BTN: без позиции играется сложнее.')
   756→    } else {
   757→      const rp = raiser && raiser !== pos ? raiser : DEFAULT_RAISER[pos] ?? 'CO'
   758→      const chart = omahaPreflop(game as 'omaha4' | 'omaha5', pos, players, 'vsraise', callers, rp)
   759→      notes.push(`Против открытия с ${rp} за столом ${players}-max: 3бет ≈ топ ${chart.threeBetPercent.toFixed(0)}%, колл ≈ топ ${chart.callPercent.toFixed(0)}%.`)
