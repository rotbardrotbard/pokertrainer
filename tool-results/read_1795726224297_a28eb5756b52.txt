    79→    79→/**
    80→    80→ * Чарт префлопа для позиции, размера стола и ситуации.
    81→    81→ * В ситуации «против рейза» ширина защиты зависит от позиции рейзера:
    82→    82→ * BB против BTN/SB защищается в 1,5–2 раза шире, чем против UTG.
    83→    83→ */
    84→    84→export function holdemPreflop(pos: Position, players: number, situation: 'open' | 'vsraise', callers: number, raiser?: Position): HoldemPreflop {
    85→    85→  if (situation === 'open') {
    86→    86→    const set = openRangeClassesFor(pos, players)
    87→    87→    return { raise: set, call: new Set(), raisePct: rangePercent(set), callPct: 0, mode: 'open' }
    88→    88→  }
    89→    89→  const rp = raiser && raiser !== pos ? raiser : DEFAULT_RAISER[pos] ?? 'CO'
    90→    90→  const multiFactor = Math.max(0.45, 1 - 0.1 * callers)
    91→    91→  const chart = vsRaiseCharts(pos, rp, players, multiFactor)
    92→    92→  if (!chart) {
    93→    93→    return { raise: new Set(), call: new Set(), raisePct: 0, callPct: 0, mode: 'vsraise' }
    94→    94→  }
    95→    95→  return { raise: chart.threeBet, call: chart.call, raisePct: chart.threeBetPct, callPct: chart.callPct, mode: 'vsraise' }
    96→    96→}

    99→    99→  if (chart.raise.has(cls)) return 'raise'
   100→   100→  if (chart.call.has(cls)) return 'call'
   101→   101→  return 'fold'
   102→   102→}

   163→   163→export function omahaPreflop(game: 'omaha4' | 'omaha5', pos: Position, players: number, situation: 'open' | 'vsraise', callers: number, raiser?: Position): OmahaPreflop {
   164→   164→  const sizeFactor = SIZE_DEF_FACTOR[players] ?? 1
   165→   165→  if (situation === 'open') {
   166→   166→    const percent = clamp(PLO_OPEN_SIZE[game][players]?.[pos] ?? PLO_OPEN_SIZE[game][6][pos], 5, 90)
   167→   167→    return {
   168→   168→      mode: 'open',
   169→   169→      openPercent: percent,
   170→   170→      threeBetPercent: 0,
   171→   171→      callPercent: 0,
   172→   172→      openThreshold: omahaRangeThreshold(game, percent),
   173→   173→      threeBetThreshold: 0,
   174→   174→      callThreshold: 0,
   175→   175→    }
   176→   176→  }
   177→   177→  const rp = raiser && raiser !== pos ? raiser : DEFAULT_RAISER[pos] ?? 'CO'
   178→   178→  const rf = RAISER_FACTOR[rp]
   179→   179→  const multiFactor = Math.max(0.45, 1 - 0.1 * callers)
   180→   180→  const bbBonus = pos === 'BB' ? 1.5 : 1
   181→   181→  const base = PLO_VSRAISE_POS[game][pos]
   182→   182→  const threeBetPercent = clamp(base.threeBet * (1 + (rf - 1) / 2) * sizeFactor, 3, 18)
   183→   183→  const callPercent = clamp(base.call * rf * sizeFactor * multiFactor * bbBonus, 5, 62)
   184→   184→  return {
   185→   185→    mode: 'vsraise',
   186→   186→    openPercent: 0,
   187→   187→    threeBetPercent,
   188→   188→    callPercent,
   189→   189→    openThreshold: 0,
   190→   190→    threeBetThreshold: omahaRangeThreshold(game, threeBetPercent),
   191→   191→    callThreshold: omahaRangeThreshold(game, callPercent),
   192→   192→  }
   193→   193→}

   195→   195→export function omahaAction(game: 'omaha4' | 'omaha5', hand: Card[], chart: OmahaPreflop, situation: 'open' | 'vsraise'): 'raise' | 'call' | 'fold' {
   196→   196→  if (situation === 'open') {
   197→   197→    return handInRange(game, hand, chart.openPercent) ? 'raise' : 'fold'
   198→   198→  }
   199→   199→  if (handInRange(game, hand, chart.threeBetPercent)) return 'raise'
   200→   200→  if (handInRange(game, hand, chart.callPercent)) return 'call'
   201→   201→  return 'fold'
   202→   202→}

   220→   220→/** Анализ текстуры доски: масть, спаренность, связанность, высота. */
   221→   221→export function analyzeBoardTexture(board: Card[]): BoardTexture {
   222→   222→  if (board.length < 3) {
   223→   223→    return {
   224→   224→      label: 'Текстура не задана', note: 'Выберите предустановку доски, чтобы учесть текстуру.',
   225→   225→      paired: false, trips: false, monotone: false, twoTone: false, rainbow: true,
   226→   226→      connected: false, straightPossible: false, high: false, empty: true,
   227→   227→    }
   228→   228→  }
   229→   229→  const rankCnt = new Map<number, number>()
   230→   230→  const suitCnt = [0, 0, 0, 0]
   231→   231→  for (const c of board) {
   232→   232→    rankCnt.set(c.r, (rankCnt.get(c.r) || 0) + 1)
   233→   233→    suitCnt[c.s]++
   234→   234→  }
   235→   235→  const counts = [...rankCnt.values()]
   236→   236→  const paired = counts.some((v) => v >= 2)
   237→   237→  const trips = counts.some((v) => v >= 3)
   238→   238→  const monotone = board.every((c) => c.s === board[0].s)
   239→   239→  const distinctSuits = suitCnt.filter((n) => n > 0).length
   240→   240→  const twoTone = !monotone && distinctSuits === 2
   241→   241→  const rainbow = !monotone && !twoTone
   242→   242→  const distinct = [...rankCnt.keys()].sort((a, b) => a - b)
   243→   243→  const extended = [...distinct]
   244→   244→  if (rankCnt.has(14)) extended.push(1)
   245→   245→  const seq = [...new Set(extended)].sort((a, b) => a - b)
   246→   246→  let straightPossible = false
   247→   247→  for (let i = 0; i + 2 < seq.length; i++) {
   248→   248→    if (seq[i + 2] - seq[i] === 2) straightPossible = true
   249→   249→  }
   250→   250→  const span = distinct[distinct.length - 1] - distinct[0]
   251→   251→  const connected = distinct.length >= 3 && span <= 5 && !paired
   252→   252→  const high = Math.max(...distinct) >= 12
   253→   253→
   254→   254→  const parts: string[] = []
   255→   255→  if (monotone) parts.push('моно-масть')
   256→   256→  else if (twoTone) parts.push('двухмастная')
   257→   257→  else parts.push('радуга')
   258→   258→  if (paired) parts.push(trips ? 'спаренная' : 'спаренная')
   259→   259→  if (straightPossible) parts.push('стрит-возможен')
   260→   260→  const label = parts.join(' · ')
   261→   261→
   262→   262→  const notes: string[] = []
   263→   263→  if (monotone) notes.push('одна масть на доске: блефам нечего усиливать, играют только натс-флеши и руки с тузом масти')
   264→   264→  else if (twoTone) notes.push('масть на доске: флеш-дро живы, полу-блефы сохраняют эквити')
   265→   265→  else notes.push('без масти: блефы работают лучше, слабые готовые руки коллируют реже')
   266→   266→  if (paired) notes.push('пара на доске: у оппонентов меньше комбинаций, ставки можно делать дешевле')
   267→   267→  if (straightPossible) notes.push('доска связана: сильные руки защищают банк от стрит-дро')
   268→   268→  if (!straightPossible && !paired && rainbow) notes.push('сухая доска: чаще блеф и мелкие банки, границы коллов ниже')
   269→   269→
   270→   270→  return {
   271→   271→    label,
   272→   272→    note: notes.join('; '),
   273→   273→    paired, trips, monotone, twoTone, rainbow, connected, straightPossible, high, empty: false,
   274→   274→  }
   275→   275→}

   599→   599→      ? 'на монотонной доске у блефов нет флеш-дро — блефуйте заметно реже'
   600→   600→      : 'на монотонной доске ставки мельче: рейзы может позволить только натс-флеш')
   601→   601→  }
   602→   602→  if (texture.twoTone && t.bluffTier) parts.push('у полу-блефов есть флеш-дро — бет или колл сохраняют эквити')
   603→   603→  if (texture.paired) parts.push('спаренная доска — ставки дешевле, границы коллов ниже')
   604→   604→  if (texture.straightPossible && !t.bluffTier) parts.push('связанная доска — размер ставки чуть больше для защиты от дро')
   605→   605→  if (texture.rainbow && !texture.paired && !texture.straightPossible && t.bluffTier) {
   606→   606→    parts.push(opponents <= 1 ? 'сухая доска — блеф с хорошим фолд-эквити' : 'сухая доска в мультпоте — блефы почти не работают')
   607→   607→  }
   608→   608→  if (texture.high && t.action === 'fold') parts.push('старшие карты на доске бьют вторые пары — не цепляйтесь')
   609→   609→  return parts.join('; ')
   610→   610→}

   612→   612→function planOneTier(t: PlanTierBase, texture: BoardTexture, opponents: number, facingPct: number | null): TierPlan {
   613→   613→  let betFreq = t.betFreq
   614→   614→  let size = t.size
   615→   615→  const isNut = t.foldAbove >= 999
   616→   616→
   617→   617→  // текстура доски
   618→   618→  if (texture.monotone) {
   619→   619→    size = Math.min(size, 50)
   620→   620→    if (t.bluffTier) betFreq *= 0.6
   621→   621→    else if (!isNut) betFreq *= 0.85
   622→   622→  }
   623→   623→  if (texture.paired) size = Math.min(size, 40)
   624→   624→  if (texture.straightPossible && !t.bluffTier && !isNut) size = Math.min(90, size + 10)
   625→   625→  if (texture.rainbow && !texture.paired && !texture.straightPossible && t.bluffTier) betFreq = Math.min(80, betFreq * 1.3)
   626→   626→
   627→   627→  // мультпот
   628→   628→  if (opponents >= 3) {
   629→   629→    if (isNut) betFreq = Math.max(betFreq, 85)
   630→   630→    if (t.bluffTier) betFreq *= 0.4
   631→   631→    size = Math.min(size, 66)
   632→   632→  }
   633→   633→  if (opponents >= 5 && t.bluffTier) betFreq = 0
   634→   634→  betFreq = Math.round(clamp(betFreq, 0, 95))
   635→   635→  size = Math.round(size)
   636→   636→
   637→   637→  // смешанные частоты на инициативе
   638→   638→  const mix: MixItem[] = []
   639→   639→  if (betFreq <= 0) {
   640→   640→    mix.push({ action: 'check', freq: 100, label: 'чек 100%' })
   641→   641→  } else {
   642→   642→    mix.push({ action: 'bet', freq: betFreq, label: `бет ${size}% банка — ${betFreq}%` })
   643→   643→    if (betFreq < 100) mix.push({ action: 'check', freq: 100 - betFreq, label: `чек ${100 - betFreq}%` })
   644→   644→  }
   645→   645→  const line = betFreq >= 85
   646→   646→    ? `бет ${size}% банка (изредка чек)`
   647→   647→    : betFreq <= 0
   648→   648→      ? 'чек (блефы почти не работают)'
   649→   649→      : `микс: бет ${size}% банка в ${betFreq}% случаев / чек в ${100 - betFreq}%`
   650→   650→
   651→   651→  const sizeParts: string[] = []
   652→   652→  if (texture.monotone) sizeParts.push('на монотонной доске не крупнее 50% банка')
   653→   653→  if (texture.paired) sizeParts.push('спаренная доска — размер поменьше')
   654→   654→  if (texture.straightPossible && !t.bluffTier && !isNut) sizeParts.push('связанная доска — размер побольше для защиты')
   655→   655→  const sizeNote = sizeParts.length > 0 ? `Размер: ${size}% банка (${sizeParts.join('; ')}).` : `Размер: ${size}% банка.`
   656→   656→
   657→   657→  // границы колл/фолд против выбранного размера ставки
   658→   658→  const multiFactor = opponents >= 4 ? 0.7 : opponents === 3 ? 0.85 : 1
   659→   659→  const callUpTo = Math.round(t.callUpTo * multiFactor)
   660→   660→  let verdict: FacingVerdict | null = null
   661→   661→  if (facingPct != null) {
   662→   662→    if (isNut) {
   663→   663→      verdict = { verdict: 'raise', note: `${t.faceText} — рейз при любой ставке оппонента` }
   664→   664→    } else if (facingPct <= callUpTo && facingPct <= t.foldAbove) {
   665→   665→      verdict = { verdict: 'call', note: `колл: ставка ${facingPct}% банка в пределах границы ~${callUpTo}% сходится по оддсам` }
   666→   666→    } else if (facingPct > t.foldAbove) {
   667→   667→      verdict = { verdict: 'fold', note: `фолд: ставка ${facingPct}% банка выше границы ~${t.foldAbove > 900 ? 'любой' : `${t.foldAbove}%`} — не сходится` }
   668→   668→    } else {
   669→   669→      verdict = { verdict: 'mix', note: `пограничная ставка: микс колл/фолд${opponents >= 2 ? ', в мультпоте чаще фолд' : ''}` }
   670→   670→    }
   671→   671→  }
   672→   672→  const vsBetNote = `колл до ~${callUpTo}% банка${t.foldAbove < 999 ? `, против ~${t.foldAbove}%+ — фолд` : ', вместо фолда — рейз'}`
   673→   673→
   674→   674→  return {
   675→   675→    tier: t.tier,
   676→   676→    examples: t.examples,
   677→   677→    action: t.action,
   678→   678→    line,
   679→   679→    mix,
   680→   680→    sizeNote,
   681→   681→    vsBetNote,
   682→   682→    verdict,
   683→   683→    textureNote: tierTextureNote(t, texture, opponents),
   684→   684→  }
   685→   685→}

   703→   703→    : 'Особенность Холдема: одна пара часто достаточна против 1 оппонента, но слаба в мультпотах.'

   707→   707→    positionNote: positionNote(seat),

   710→   710→    textureNote: texture.note,
   711→   711→    facingLabel: facingPct != null ? `Оппонент поставил ${facingPct}% банка` : null,

   714→   714→}

   716→   716→/**
   717→   717→ * Ориентир из GTO-чарта для Омахи по вердикту решения:
   718→   718→ * raise → верхний тир (натсы), call → средний, fold → нижний.
   719→   719→ * Используется симулятором раздач для связи с разделом GTO.
   720→   720→ */
   721→   721→export function omahaEquityTier(game: 'omaha4' | 'omaha5', street: Exclude<Street, 'preflop'>, verdict: 'raise' | 'call' | 'fold'): PostflopTier {
   722→   722→  const tiers = GUIDES[`${game}:${street}`] ?? OMAHA_FLOP
   723→   723→  const pick = verdict === 'raise' ? tiers[0] : verdict === 'call' ? tiers[1] : tiers[tiers.length - 1]
   724→   724→  return { tier: pick.tier, examples: pick.examples, aggressive: pick.aggrText, facingBet: pick.faceText, action: pick.action }
   725→   725→}

   729→   729→export function preflopNotes(game: GameKind, pos: Position, situation: 'open' | 'vsraise', callers: number, raiser?: Position, players = 6): string[] {
   730→   730→  const notes: string[] = []
   731→   731→  if (game === 'holdem') {
   732→   732→    if (situation === 'open') {
   733→   733→      notes.push(`Стол ${players}-max: открытие с ${pos} — топ ~${openPercentFor(pos, players).toFixed(0)}% рук; отдельный чарт для каждого размера стола.`)
   734→   734→      notes.push('Открытие рейзом 2–3bb: чем раньше позиция, тем тайтовее диапазон.')
   735→   735→      if (pos === 'BTN') notes.push('С баттона открывается самый широкий диапазон — вы играете в позиции на всех блайндах.')
   736→   736→      if (pos === 'UTG') notes.push('С UTG после вас действует много игроков: нужна сила руки, а не потенциал.')
   737→   737→      if (pos === 'SB') notes.push('С SB против одного BB можно открывать шире, но без позиции постфлоп сложнее.')
   738→   738→    } else {
   739→   739→      const rp = raiser && raiser !== pos ? raiser : DEFAULT_RAISER[pos] ?? 'CO'
   740→   740→      const chart = holdemPreflop(pos, players, 'vsraise', callers, rp)
   741→   741→      notes.push(`Против открытия с ${rp} за столом ${players}-max: 3бет ≈ топ ${chart.raisePct.toFixed(0)}% рук, колл ≈ топ ${chart.callPct.toFixed(0)}%.`)
   742→   742→      notes.push('Ширина защиты зависит от позиции рейзера: BB против BTN/SB коллирует примерно в 1,5–2 раза больше рук, чем против UTG.')
   743→   743→      notes.push('Против рейза: премиум — 3бет, играбельные — колл, остальное — фолд.')
   744→   744→      if (callers > 0) notes.push(`Рейз уже поддержали ${callers} ${callers === 1 ? 'игрок' : 'игрока'}: коллы требуют более сильных рук, мультпот — для натс-потенциала.`)
   745→   745→      if (pos === 'BB') notes.push('BB защищается шире всех — колл со скидкой блайнда.')
   746→   746→      if (pos === 'BTN') notes.push('С баттона против рейза много коллов в позиции и 3беты с премиумом.')
   747→   747→    }
   748→   748→  } else {
   749→   749→    if (situation === 'open') {
   750→   750→      const pct = omahaPreflop(game as 'omaha4' | 'omaha5', pos, players, 'open', 0).openPercent
   751→   751→      notes.push(`Стол ${players}-max: открытие с ${pos} — топ ~${pct.toFixed(0)}% рук; отдельный чарт для каждого размера стола.`)
   752→   752→      notes.push('В Омахе открывайте координированные руки: тузы, мастевость и связки вместе сильнее, чем по отдельности.')
   753→   753→      if (pos === 'UTG') notes.push('С UTG — тайтово: двойномастные тузы, бродвей-связки, высокие пары.')
   754→   754→      if (pos === 'BTN') notes.push('С баттона — самый широкий диапазон: все двойномастные, связки, пары и A-хай руки.')
   755→   755→      if (pos === 'SB') notes.push('С SB открывайте чуть тайтовее BTN: без позиции играется сложнее.')
   756→   756→    } else {
   757→   757→      const rp = raiser && raiser !== pos ? raiser : DEFAULT_RAISER[pos] ?? 'CO'
   758→   758→      const chart = omahaPreflop(game as 'omaha4' | 'omaha5', pos, players, 'vsraise', callers, rp)
   759→   759→      notes.push(`Против открытия с ${rp} за столом ${players}-max: 3бет ≈ топ ${chart.threeBetPercent.toFixed(0)}%, колл ≈ топ ${chart.callPercent.toFixed(0)}%.`)
   760→   760→      notes.push('Ширина защиты зависит от позиции рейзера: против поздних открытий (BTN/SB) коллы в 1,5–1,7 раза шире, чем против UTG.')

   762→   762→      if (callers > 0) notes.push(`В банке уже ${callers + 1} ${callers + 1 === 1 ? 'игрок' : 'игроков'}: коллируйте руки с натс-потенциалом, маргинальные — фолд.`)
   763→   763→      if (pos === 'BB') notes.push('BB защищается шире всех: скидка блайнда делает колл прибыльным.')
   764→   764→      notes.push('Изолированные тузы (AAxx без мастей/связок) в Омахе — рука «на один постфлоп»: не переоценивайте их.')
   765→   765→    }
   766→   766→  }
   767→   767→  return notes
   768→   768→}
